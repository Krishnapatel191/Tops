STEP 1: BUILD WITH AI
#include <stdio.h>

int main()
{
    int arr[10];
    int i, j, temp;
    int max, min;
    float sum = 0, mean;

    printf("Enter 10 integers:\n");

    for(i = 0; i < 10; i++)
    {
        scanf("%d", &arr[i]);
        sum = sum + arr[i];
    }

    max = arr[0];
    min = arr[0];

    for(i = 1; i < 10; i++)
    {
        if(arr[i] > max)
            max = arr[i];

        if(arr[i] < min)
            min = arr[i];
    }

    mean = sum / 10;

    printf("\nMaximum = %d", max);
    printf("\nMinimum = %d", min);
    printf("\nMean = %.2f", mean);


    // Bubble Sort
    for(i = 0; i < 10; i++)
    {
        for(j = 0; j < 9; j++)
        {
            if(arr[j] > arr[j+1])
            {
                temp = arr[j];
                arr[j] = arr[j+1];
                arr[j+1] = temp;
            }
        }
    }


    printf("\nSorted Array: ");

    for(i = 0; i < 10; i++)
    {
        printf("%d ", arr[i]);
    }


    // Mean comparison
    if(mean - min < max - mean)
        printf("\nMean is closer to minimum");

    else if(max - mean < mean - min)
        printf("\nMean is closer to maximum");

    else
        printf("\nMean is exactly midway between minimum and maximum");


    return 0;
}

{

    min = arr[0];

    for(i = 1; i < 10; i++)
    {

            max = arr[i];

        if(arr[i] < min)
            min = arr[i];
    }


    mean = sum / 10.0;


    printf("\nMaximum = %d", max);
    printf("\nMinimum = %d", min);

    // Bubble Sort
    for(i = 0; i < 9; i++)
    {
        for(j = 0; j < 9-i; j++)
        {
            if(arr[j] > arr[j+1])
            {
                temp = arr[j];
                arr[j] = arr[j+1];
                arr[j+1] = temp;
            }
        }
    }


    printf("\nSorted Array: ");

    for(i = 0; i < 10; i++)
    {
        printf("%d ", arr[i]);
    }

    maxDistance = max - mean;
